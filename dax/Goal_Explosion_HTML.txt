Goal_Explosion_HTML = 

VAR SelectedSeason =

    COALESCE(

        SELECTEDVALUE('public matches'[SEASON]),

        MAXX(

            FILTER(

                ALL('public matches'),

                'public matches'[STAGE] = "League phase"

                    && NOT ISBLANK('public matches'[HOME_TEAM_SCORE])

                    && NOT ISBLANK('public matches'[AWAY_TEAM_SCORE])

            ),

            'public matches'[SEASON]

        )

    )

-- COMPLETED GAMES

VAR Games =

    FILTER(

        ALL('public matches'),

        'public matches'[SEASON] = SelectedSeason

            && 'public matches'[STAGE] = "League phase"

            && NOT ISBLANK('public matches'[HOME_TEAM_SCORE])

            && NOT ISBLANK('public matches'[AWAY_TEAM_SCORE])

            && NOT ISBLANK('public matches'[MATCH_DATE])

    )

-- EACH GAME'S ROUND NUMBER

VAR GamesWithRound =

    ADDCOLUMNS(

        Games,

        "@Round",

            VAR CurrentTeam = 'public matches'[HOME_TEAM]

            VAR CurrentDate = 'public matches'[MATCH_DATE]

            VAR CurrentID = 'public matches'[MATCH_ID]

            RETURN

                COUNTROWS(

                    FILTER(

                        Games,

                        (

                            'public matches'[HOME_TEAM] = CurrentTeam

                            || 'public matches'[AWAY_TEAM] = CurrentTeam

                        )

                        &&

                        (

                            'public matches'[MATCH_DATE] < CurrentDate

                            ||

                            (

                                'public matches'[MATCH_DATE] = CurrentDate

                                && 'public matches'[MATCH_ID] <= CurrentID

                            )

                        )

                    )

                ),

        "@Goals",

            'public matches'[HOME_TEAM_SCORE] + 'public matches'[AWAY_TEAM_SCORE]

    )

-- MATCH ROUNDS

VAR Rounds =

    SELECTCOLUMNS(

        GENERATESERIES(1, 8, 1),

        "RoundNo", [Value]

    )

VAR RoundStats =

    ADDCOLUMNS(

        Rounds,

        "@Goals",

            VAR R = [RoundNo]

            RETURN

                COALESCE(

                    SUMX(

                        FILTER(

                            GamesWithRound,

                            [@Round] = R

                        ),

                        [@Goals]

                    ),

                    0

                ),

        "@Matches",

            VAR R = [RoundNo]

            RETURN

                COUNTROWS(

                    FILTER(

                        GamesWithRound,

                        [@Round] = R

                    )

                )

    )

VAR ActiveRounds =

    FILTER(

        RoundStats,

        [@Matches] > 0

    )

VAR MaxGoals =

    COALESCE(MAXX(ActiveRounds, [@Goals]), 0)

VAR MinGoals =

    COALESCE(MINX(ActiveRounds, [@Goals]), 0)

VAR TotalGoals =

    COALESCE(

        SUMX(

            Games,

            'public matches'[HOME_TEAM_SCORE] + 'public matches'[AWAY_TEAM_SCORE]

        ),

        0

    )

VAR PeakRound =

    MINX(

        FILTER(ActiveRounds, [@Goals] = MaxGoals),

        [RoundNo]

    )

VAR LowRound =

    MINX(

        FILTER(ActiveRounds, [@Goals] = MinGoals),

        [RoundNo]

    )

-- CLUBS

VAR Clubs =

    DISTINCT(

        UNION(

            SELECTCOLUMNS(

                Games,

                "Club", 'public matches'[HOME_TEAM]

            ),

            SELECTCOLUMNS(

                Games,

                "Club", 'public matches'[AWAY_TEAM]

            )

        )

    )

-- CLUB GOALS + LOGOS

VAR ClubStats =
    ADDCOLUMNS(
        Clubs,
        "@GF",
            VAR T = [Club]
            RETURN COALESCE(
                SUMX(
                    Games,
                    SWITCH(
                        TRUE(),
                        'public matches'[HOME_TEAM] = T, 'public matches'[HOME_TEAM_SCORE],
                        'public matches'[AWAY_TEAM] = T, 'public matches'[AWAY_TEAM_SCORE],
                        0
                    )
                ), 0
            ),
        "@GA",
            VAR T = [Club]
            RETURN COALESCE(
                SUMX(
                    Games,
                    SWITCH(
                        TRUE(),
                        'public matches'[HOME_TEAM] = T, 'public matches'[AWAY_TEAM_SCORE],
                        'public matches'[AWAY_TEAM] = T, 'public matches'[HOME_TEAM_SCORE],
                        0
                    )
                ), 0
            ),
        "@Logo",
            VAR T = [Club]
            RETURN MAXX(
                FILTER(ALL('public teams'), 'public teams'[TEAM_NAME] = T),
                'public teams'[TEAM_URL]
            )
    )

VAR ClubStatsWithGD =
    ADDCOLUMNS(ClubStats, "@GD", [@GF] - [@GA])

-- MOST GOALS CLUB

VAR TopClub =

    TOPN(

        1,

        ClubStatsWithGD,

        [@GF], DESC,
        [@GD], DESC,
        [@GA], ASC,
        [Club], ASC

    )

VAR TopName =

    COALESCE(MAXX(TopClub, [Club]), "-")

VAR TopGoals =

    COALESCE(MAXX(TopClub, [@GF]), 0)

VAR TopLogo =

    MAXX(TopClub, [@Logo])

-- FEWEST GOALS CLUB

VAR LowClub =

    TOPN(

        1,

        ClubStatsWithGD,

        [@GF], ASC,
        [@GD], ASC,
        [@GA], DESC,
        [Club], ASC

    )

VAR LowName =

    COALESCE(MAXX(LowClub, [Club]), "-")

VAR LowGoals =

    COALESCE(MAXX(LowClub, [@GF]), 0)

VAR LowLogo =

    MAXX(LowClub, [@Logo])

-- SAFE NAMES FOR HTML

VAR TopSafeName =

    SUBSTITUTE(

        SUBSTITUTE(

            SUBSTITUTE(

                SUBSTITUTE(TopName, "&", "&"),

                "<", "<"

            ),

            ">", ">"

        ),

        "'", "'"

    )

VAR LowSafeName =

    SUBSTITUTE(

        SUBSTITUTE(

            SUBSTITUTE(

                SUBSTITUTE(LowName, "&", "&"),

                "<", "<"

            ),

            ">", ">"

        ),

        "'", "'"

    )

-- MATCH BARS

VAR Bars =

    CONCATENATEX(

        RoundStats,

        VAR R = [RoundNo]

        VAR Goals = [@Goals]

        VAR MatchCount = [@Matches]

        VAR BarPercent =

            DIVIDE(Goals, MaxGoals, 0) * 100

        VAR BarColor =

            SWITCH(

                TRUE(),

                MatchCount = 0, "#20345F",

                MaxGoals > MinGoals && Goals = MaxGoals, "#FF405B",

                MaxGoals > MinGoals && Goals = MinGoals, "#FFD34D",

                "#00BFFF"

            )

        VAR GoalColor =

            SWITCH(

                TRUE(),

                MatchCount = 0, "#7890BC",

                MaxGoals > MinGoals && Goals = MaxGoals, "#FF6678",

                MaxGoals > MinGoals && Goals = MinGoals, "#FFD34D",

                "#FFFFFF"

            )

        RETURN

            "<div style='

                display:flex;

                align-items:center;

                gap:7px;

                width:100%;

                height:30px;

                box-sizing:border-box;

            '>"

            & "<span style='

                width:22px;

                flex:0 0 22px;

                font-size:11px;

                font-weight:700;

                color:#AFC8FF;

            '>M"

            & FORMAT(R, "0")

            & "</span>"

            & "<div style='

                flex:1;

                min-width:0;

                height:13px;

                background:#1B3268;

                border-radius:7px;

                overflow:hidden;

            '>"

            & "<div style='

                width:"

            & FORMAT(BarPercent, "0.0", "en-US")

            & "%;

                height:100%;

                background:"

            & BarColor

            & ";

                border-radius:7px;

            '></div>"

            & "</div>"

            & "<span style='

                width:28px;

                flex:0 0 28px;

                text-align:right;

                font-size:13px;

                font-weight:800;

                color:"

            & GoalColor

            & ";

            '>"

            & IF(

                MatchCount = 0,

                "-",

                FORMAT(Goals, "0")

            )

            & "</span></div>",

        "",

        [RoundNo], ASC

    )

-- LOGOS

VAR TopLogoHTML =

    IF(

        NOT ISBLANK(TopLogo) && TopLogo <> "",

        "<img src='" & TopLogo & "'

            style='width:33px;height:33px;

        "<span style='font-size:23px;'>РІС™Р…</span>"

        width:36px;

        display:flex;

        align-items:center;

        justify-content:center;

    '>"

    & TopLogoHTML

    & "</div>"

    & "<div style='

        word-break:normal;

    '>"

-- FEWEST GOALS CARD

VAR LowCard =

    "<div style='

        display:flex;

        align-items:center;

        gap:7px;

        width:100%;

        box-sizing:border-box;

        padding:9px 7px;

        background:#112A60;

        border-left:3px solid #FFD34D;

        border-radius:7px;

    '>"

    & "<div style='

        width:36px;

        flex:0 0 36px;

        display:flex;

        align-items:center;

        justify-content:center;

    '>"

    & LowLogoHTML

    & "</div>"

    & "<div style='

        flex:1;

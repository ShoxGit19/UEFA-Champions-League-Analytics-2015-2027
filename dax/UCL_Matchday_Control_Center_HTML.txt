UCL_Matchday_Control_Center_HTML = 
VAR SeasonSelected =
    COALESCE (
        SELECTEDVALUE ( 'public matches'[SEASON] ),
        "2024-2025"
    )

VAR SeasonLabel =
    SUBSTITUTE ( SeasonSelected, "-", "/" )

VAR SelectedTeam =
    SELECTEDVALUE ( 'public teams'[TEAM_NAME] )

VAR SelectedCountry =
    SELECTEDVALUE ( 'public teams'[COUNTRY] )

VAR BaseMatches =
    FILTER (
        ALLSELECTED ( 'public matches' ),
        'public matches'[SEASON] = SeasonSelected
            && 'public matches'[STAGE] = "League phase"
            && NOT ISBLANK ( 'public matches'[MATCH_ID] )
            && NOT ISBLANK ( 'public matches'[HOME_TEAM] )
            && NOT ISBLANK ( 'public matches'[AWAY_TEAM] )
            && NOT ISBLANK ( 'public matches'[DATE_TIME] )
            && (
                ISBLANK ( SelectedTeam )
                    || 'public matches'[HOME_TEAM] = SelectedTeam
                    || 'public matches'[AWAY_TEAM] = SelectedTeam
            )
            && (
                ISBLANK ( SelectedCountry )
                    || LOOKUPVALUE (
                        'public teams'[COUNTRY],
                        'public teams'[TEAM_NAME], 'public matches'[HOME_TEAM]
                    ) = SelectedCountry
                    || LOOKUPVALUE (
                        'public teams'[COUNTRY],
                        'public teams'[TEAM_NAME], 'public matches'[AWAY_TEAM]
                    ) = SelectedCountry
            )
    )

VAR MatchesWithMatchday =
    ADDCOLUMNS (
        BaseMatches,
        "__MATCHDAY",
            VAR MatchDate =
                INT ( 'public matches'[DATE_TIME] )
            RETURN
                SWITCH (
                    TRUE (),
                    MatchDate >= DATE ( 2024, 9, 17 )
                        && MatchDate <= DATE ( 2024, 9, 19 ), "MD1",
                    MatchDate >= DATE ( 2024, 10, 1 )
                        && MatchDate <= DATE ( 2024, 10, 2 ), "MD2",
                    MatchDate >= DATE ( 2024, 10, 22 )
                        && MatchDate <= DATE ( 2024, 10, 23 ), "MD3",
                    MatchDate >= DATE ( 2024, 11, 5 )
                        && MatchDate <= DATE ( 2024, 11, 6 ), "MD4",
                    MatchDate >= DATE ( 2024, 11, 26 )
                        && MatchDate <= DATE ( 2024, 11, 27 ), "MD5",
                    MatchDate >= DATE ( 2024, 12, 10 )
                        && MatchDate <= DATE ( 2024, 12, 11 ), "MD6",
                    MatchDate >= DATE ( 2025, 1, 21 )
                        && MatchDate <= DATE ( 2025, 1, 22 ), "MD7",
                    MatchDate = DATE ( 2025, 1, 29 ), "MD8",
                    "OTHER"
                )
    )

VAR VisibleMinDate =
    MINX (
        ALLSELECTED ( 'public matches' ),
        INT ( 'public matches'[DATE_TIME] )
    )

VAR VisibleMaxDate =
    MAXX (
        ALLSELECTED ( 'public matches' ),
        INT ( 'public matches'[DATE_TIME] )
    )

VAR SelectedMatchday =
    SWITCH (
        TRUE (),
        VisibleMinDate >= DATE ( 2024, 9, 17 )
            && VisibleMaxDate <= DATE ( 2024, 9, 19 ), "MD1",
        VisibleMinDate >= DATE ( 2024, 10, 1 )
            && VisibleMaxDate <= DATE ( 2024, 10, 2 ), "MD2",
        VisibleMinDate >= DATE ( 2024, 10, 22 )
            && VisibleMaxDate <= DATE ( 2024, 10, 23 ), "MD3",
        VisibleMinDate >= DATE ( 2024, 11, 5 )
            && VisibleMaxDate <= DATE ( 2024, 11, 6 ), "MD4",
        VisibleMinDate >= DATE ( 2024, 11, 26 )
            && VisibleMaxDate <= DATE ( 2024, 11, 27 ), "MD5",
        VisibleMinDate >= DATE ( 2024, 12, 10 )
            && VisibleMaxDate <= DATE ( 2024, 12, 11 ), "MD6",
        VisibleMinDate >= DATE ( 2025, 1, 21 )
            && VisibleMaxDate <= DATE ( 2025, 1, 22 ), "MD7",
        VisibleMinDate = DATE ( 2025, 1, 29 )
            && VisibleMaxDate = DATE ( 2025, 1, 29 ), "MD8",
        "MD1"
    )

VAR DisplayMatches =
    FILTER (
        MatchesWithMatchday,
        [__MATCHDAY] = SelectedMatchday
    )

VAR MatchCount =
    COUNTROWS ( DisplayMatches )

VAR TotalGoals =
    SUMX (
        DisplayMatches,
        COALESCE ( 'public matches'[HOME_TEAM_SCORE], 0 )
            + COALESCE ( 'public matches'[AWAY_TEAM_SCORE], 0 )
    )

VAR TotalAttendance =
    SUMX (
        DisplayMatches,
        COALESCE ( 'public matches'[ATTENDANCE], 0 )
    )

VAR AverageGoals =
    DIVIDE ( TotalGoals, MatchCount )

VAR SelectedMatchID =
    SELECTEDVALUE ( 'public matches'[MATCH_ID] )

VAR DetailMatch =
    TOPN (
        1,
        FILTER (
            DisplayMatches,
            ISBLANK ( SelectedMatchID )
                || 'public matches'[MATCH_ID] = SelectedMatchID
        ),
        'public matches'[DATE_TIME], ASC,
        'public matches'[MATCH_ID], ASC
    )

VAR DetailMatchID =
    MAXX ( DetailMatch, 'public matches'[MATCH_ID] )

VAR DetailHomeTeam =
    MAXX ( DetailMatch, 'public matches'[HOME_TEAM] )

VAR DetailAwayTeam =
    MAXX ( DetailMatch, 'public matches'[AWAY_TEAM] )

VAR DetailHomeScore =
    MAXX ( DetailMatch, 'public matches'[HOME_TEAM_SCORE] )

VAR DetailAwayScore =
    MAXX ( DetailMatch, 'public matches'[AWAY_TEAM_SCORE] )

VAR DetailDateTime =
    MAXX ( DetailMatch, 'public matches'[DATE_TIME] )

VAR DetailStadium =
    MAXX ( DetailMatch, 'public matches'[STADIUM] )

VAR DetailAttendance =
    MAXX ( DetailMatch, 'public matches'[ATTENDANCE] )

VAR DetailHomeLogo =
    LOOKUPVALUE (
        'public teams'[TEAM_URL],
        'public teams'[TEAM_NAME], DetailHomeTeam
    )

VAR DetailAwayLogo =
    LOOKUPVALUE (
        'public teams'[TEAM_URL],
        'public teams'[TEAM_NAME], DetailAwayTeam
    )

VAR DetailCapacity =
    LOOKUPVALUE (
        'public stadiums'[CAPACITY],
        'public stadiums'[NAME], DetailStadium
    )

VAR GoalRows =
    FILTER (
        ALL ( 'public goals' ),
        'public goals'[MATCH_ID] = DetailMatchID
            && 'public goals'[SEASON] = SeasonSelected
    )

VAR GoalRowsWithPlayers =
    ADDCOLUMNS (
        GoalRows,
        "__SCORER_NAME",
            LOOKUPVALUE (
                'public players'[FULL_NAME],
                'public players'[PLAYER_ID], 'public goals'[PID]
            ),
        "__ASSIST_NAME",
            LOOKUPVALUE (
                'public players'[FULL_NAME],
                'public players'[PLAYER_ID], 'public goals'[ASSIST]
            ),
        "__SCORER_TEAM",
            LOOKUPVALUE (
                'public players'[TEAM],
                'public players'[PLAYER_ID], 'public goals'[PID]
            )
    )

VAR GoalRowsEnriched =
    ADDCOLUMNS (
        GoalRowsWithPlayers,
        "__GOAL_TEAM",
            VAR ScorerTeam = [__SCORER_TEAM]
            VAR GoalDesc =
                LOWER ( COALESCE ( 'public goals'[GOAL_DESC], "" ) )
            VAR IsOwnGoal =
                CONTAINSSTRING ( GoalDesc, "own" )
                    || CONTAINSSTRING ( GoalDesc, "autogol" )
            RETURN
                IF (
                    IsOwnGoal,
                    IF (
                        ScorerTeam = DetailHomeTeam,
                        DetailAwayTeam,
                        IF (
                            ScorerTeam = DetailAwayTeam,
                            DetailHomeTeam,
                            BLANK ()
                        )
                    ),
                    ScorerTeam
                ),
        "__TYPE_LABEL",
            VAR GoalDesc =
                LOWER ( COALESCE ( 'public goals'[GOAL_DESC], "" ) )
            RETURN
                SWITCH (
                    TRUE (),
                    CONTAINSSTRING ( GoalDesc, "own" )
                        || CONTAINSSTRING ( GoalDesc, "autogol" ), "OG",
                    CONTAINSSTRING ( GoalDesc, "pen" ), "PEN",
                    ""
                )
    )

VAR HomeGoals =
    FILTER (
        GoalRowsEnriched,
        [__GOAL_TEAM] = DetailHomeTeam
    )

VAR AwayGoals =
    FILTER (
        GoalRowsEnriched,
        [__GOAL_TEAM] = DetailAwayTeam
    )

VAR HomeGoalsHTML =
    CONCATENATEX (
        HomeGoals,
        VAR Scorer =
            COALESCE ( [__SCORER_NAME], "Unknown" )
        VAR AssistName =
            [__ASSIST_NAME]
        VAR AssistText =
            IF (
                ISBLANK ( AssistName )
                    || AssistName = "",
                "Assist: вЂ”",
                "Assist: " & AssistName
            )
        VAR GoalType =
            IF (
                [__TYPE_LABEL] = "",
                "",
                " (" & [__TYPE_LABEL] & ")"
            )
        RETURN
            "<div style='display:flex;gap:8px;align-items:flex-start;padding:7px 4px;border-bottom:1px solid rgba(175,200,255,.14);font-size:12px;'>"
                & "<span style='min-width:38px;color:#00D9FF;font-weight:800;'>"
                & FORMAT ( 'public goals'[DURATION], "0" )
                & "'</span>"
                & "<span style='color:#FFFFFF;'>вљЅ</span>"
                & "<span style='flex:1;color:#FFFFFF;'>"
                & Scorer
                & GoalType
                & "<br><span style='font-size:10px;color:#9CB8E8;'>"
                & AssistText
                & "</span></span></div>",
        "",
        'public goals'[DURATION], ASC,
        'public goals'[GOAL_ID], ASC
    )

VAR AwayGoalsHTML =
    CONCATENATEX (
        AwayGoals,
        VAR Scorer =
            COALESCE ( [__SCORER_NAME], "Unknown" )
        VAR AssistName =
            [__ASSIST_NAME]
        VAR AssistText =
            IF (
                ISBLANK ( AssistName )
                    || AssistName = "",
                "Assist: вЂ”",
                "Assist: " & AssistName
            )
        VAR GoalType =
            IF (
                [__TYPE_LABEL] = "",
                "",
                " (" & [__TYPE_LABEL] & ")"
            )
        RETURN
            "<div style='display:flex;gap:8px;align-items:flex-start;padding:7px 4px;border-bottom:1px solid rgba(175,200,255,.14);font-size:12px;'>"
                & "<span style='min-width:38px;color:#FF4F68;font-weight:800;'>"
                & FORMAT ( 'public goals'[DURATION], "0" )
                & "'</span>"
                & "<span style='color:#FFFFFF;'>вљЅ</span>"
                & "<span style='flex:1;color:#FFFFFF;'>"
                & Scorer
                & GoalType
                & "<br><span style='font-size:10px;color:#9CB8E8;'>"
                & AssistText
                & "</span></span></div>",
        "",
        'public goals'[DURATION], ASC,
        'public goals'[GOAL_ID], ASC
    )

VAR HomeGoalsFinal =
    IF (
        HomeGoalsHTML = "",
        "<div style='padding:14px;color:#9CB8E8;'>Gol qayd etilmagan</div>",
        HomeGoalsHTML

                & RowColor
                & ";border-bottom:1px solid rgba(175,200,255,.14);font-family:Segoe UI,Arial,sans-serif;'>"
                & "<div style='font-size:10px;line-height:13px;color:#AFC8FF;'>"
                & FORMAT ( MatchDateTime, "dd MMM yyyy" )
                & "<br>"
                & FORMAT ( MatchDateTime, "HH:mm" )
                & "</div>"
                & "<div style='text-align:center;'>"
                & HomeLogoHTML
                & "</div>"
                & "<div style='font-size:11px;font-weight:700;color:#FFFFFF;overflow-wrap:anywhere;'>"
                & HomeTeam
                & "</div>"
                & "<div style='text-align:center;color:#FFFFFF;font-size:16px;font-weight:900;'>"
                & FORMAT ( HomeScore, "0" )
                & " - "
                & FORMAT ( AwayScore, "0" )
                & "</div>"
                & "<div style='font-size:11px;font-weight:700;color:#FFFFFF;overflow-wrap:anywhere;'>"
                & AwayTeam
                & "</div>"
                & "<div style='text-align:center;'>"
                & AwayLogoHTML
                & "</div>"
                & "<div style='font-size:9px;line-height:12px;color:#9CB8E8;text-align:right;overflow-wrap:anywhere;'>"
                & StadiumName
                & "<br>Capacity: "
                & IF (
                    ISBLANK ( Capacity ),
                    "N/A",
                    FORMAT ( Capacity, "#,##0" )
                )
                & "</div></div>",
        "",
        'public matches'[DATE_TIME], ASC,
        'public matches'[MATCH_ID], ASC
    )

VAR MatchRowsFinal =
